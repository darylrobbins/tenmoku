\section[Main]{Main result}\label{sec:main}
\begin{theorem}\label{thm:triangle}
  For all $\vec{x}, \vec{y} \in \R^n$,
  \begin{equation}
    \norm{\vec{x} + \vec{y}} \le \norm{\vec{x}} + \norm{\vec{y}}.
  \end{equation}
\end{theorem}

\begin{proof}
  Square both sides and apply Cauchy--Schwarz:
  \[
    \lvert \langle x, y \rangle \rvert \leq \norm{x}\,\norm{y}.
  \]
\end{proof}

\begin{remark}
  Equality holds when $\vec{y} = \lambda \vec{x}$ for some $\lambda \ge 0$.
\end{remark}
